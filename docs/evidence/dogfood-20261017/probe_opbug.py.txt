#!/usr/bin/env python3
"""Prove the sanitizer's operator-stripping bug with discriminating probes.

Hypothesis (from src/persistence/stock_sqlite.rs:1085-1100, sanitize_fts5_query
steps 4-5): the dangling-boolean-operator stripper uses starts_with/ends_with on
the WHOLE query string instead of whole-token matching, so a first token that
merely begins with and/or/not, or a last token that merely ends with them, has
those characters chopped off. The query then searches a different word, finds
nothing, and returns hits=0 with no error.

Each probe reports: what the user typed, what `search` returned, and what the
unmangled FTS5 MATCH returns on the same file (ground truth for the doc).
"""
from __future__ import annotations

import os
import sqlite3
import sys
import tempfile

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from transcript_archive import Adapter  # noqa: E402

ad = sys.argv[1]
scratch = tempfile.mkdtemp(prefix="hrs-opbug-")
db = os.path.join(scratch, "state.db")

CONTENT = ("android notes and the command brand oracle order elephant "
           "nothing notable band")

with Adapter(ad) as a:
    a.open(db, create=True)
    a.ensure_session("s1", source="opbug", model="m")
    a.append_message("s1", role="user", content=CONTENT)
    a.call("checkpoint")

    probes = [
        "note",          # first token starts with NOT -> anything left?
        "notes",
        "nothing",
        "notable",
        "android",       # starts with AND
        "order",         # starts with OR
        "oracle",
        "command",       # LAST token ends with AND (only if it is the last token)
        "elephant",
        "brand",         # ends with AND as a single word
        "band",
        "command prompt",  # 'command' is first token here -> prefix arm does not fire
        "the command",     # 'command' is last -> suffix arm fires
        "and",
        "or",
        "not",
    ]
    print(f"{'typed by the user':<20} {'adapter hits':>12} {'raw MATCH (truth)':>18}  verdict")
    con = sqlite3.connect(db)
    for q in probes:
        hits = len(a.search(q, limit=20)["hits"])
        # ground truth: the plain FTS5 MATCH of exactly what the user typed
        try:
            truth = con.execute("SELECT COUNT(*) FROM messages_fts WHERE messages_fts MATCH ?",
                                (q,)).fetchone()[0]
        except sqlite3.Error as e:
            truth = f"ERR {e}"
        verdict = "ok" if (hits > 0 or truth == 0) else "SILENT MISS"
        print(f"{q:<20} {hits:>12} {str(truth):>18}  {verdict}")
    con.close()
    a.close_store()
print("scratch:", scratch)
