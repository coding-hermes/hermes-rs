#!/usr/bin/env python3
"""Capture: a SECOND backup to the same destination fails with a bare Io error."""
from __future__ import annotations

import json
import os
import shutil
import sys
import tempfile

sys.path.insert(0, "/home/kara/hrs-dogfood-20261017")
from transcript_archive import Adapter, SpecError  # noqa: E402

ad = sys.argv[1]
scratch = tempfile.mkdtemp(prefix="hrs-backup-")
db = os.path.join(scratch, "state.db")
dest = os.path.join(scratch, "state.db.backup")

with Adapter(ad) as a:
    rep = a.open(db, create=True)
    a.ensure_session("s1", source="backup", model="m")
    a.append_message("s1", role="user", content="row one")
    # a first backup (a nightly loop's first run)
    print("first  backup ->", json.dumps(a.call("backup", dest=dest)))
    print("   dest exists:", os.path.exists(dest))
    # second backup to the SAME destination (the next night's run)
    try:
        print("second backup ->", json.dumps(a.call("backup", dest=dest)))
    except SpecError as e:
        print(f"second backup -> SpecError(error={e.error!r}, cause={e.cause!r}, detail={e.detail!r})")
    a.close_store()
print("scratch:", scratch)
