#!/usr/bin/env python3
"""Append DF-HRS-111 (tracked+regenerated evidence file)."""
from __future__ import annotations

import json
import shutil
import time

BOARD = "/home/kara/hermes-rs/.coding-hermes/board/tasks.jsonl"
TS = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
RID = "DF-HRS-111"
existing = open(BOARD).read()
assert '"id":"%s"' % RID not in existing and '"id": "%s"' % RID not in existing
before = [json.loads(l)["id"] for l in existing.splitlines() if l.strip()]
shutil.copy2(BOARD, "/tmp/hrs-board-tasks.pre-dogfood-111.jsonl")

row = {
    "id": RID,
    "title": "Dogfood: a tracked evidence file is regenerated by every `cargo test` — the tree is permanently dirty after any test run (CI whitespace class)",
    "status": "pending",
    "priority": "P2",
    "labels": ["dogfood", "ci", "hygiene"],
    "complexity": 2,
    "created_at": TS,
    "updated_at": TS,
    "attempts": 0,
    "blocks": [],
    "depends_on": [],
    "capability_tags": [],
    "worker_status": "pending",
    "completed_at": None,
    "commit_hash": None,
    "foreman_note": None,
    "reasoning": None,
    "detail": (
        "tests/hrs003_python_interop.rs:23 writes the checked-in evidence file "
        "docs/evidence/hrs003/broken-case-transcript.txt on EVERY test run (the non-vacuity leg "
        "runs the driver's --mode broken), and the file embeds a wall-clock Date line plus the "
        "per-run temp adapter path. Observed for real: after a clean `cargo test --workspace` "
        "(rc=0, 29 passed) the working tree was dirty with exactly that file, diff = 2 lines "
        "(Date: 2026-10-16T14:09:43Z -> 2026-10-17T14:47:00Z; Adapter binary "
        "/var/tmp/hermes-scratch/hrs003-adapter-broken-1618961 -> ...-1527313) — and the file was "
        "ALREADY dirty when this dogfood tick started, so every lane that runs the suite leaves "
        "the same dirt for the next one. It has already cost the repo one CI cycle: commit d3877a8 "
        "'strip blank line at EOF in DF-HRS-103 control-host proof log (CI whitespace gate)'. "
        "Consequences: `git status` can never be clean on a host that ran the suite, unrelated "
        "commits capture the churn if a lane commits by wildcard, and CI gates on whitespace keep "
        "firing on a regenerated artifact. Fix direction: keep the tracked file as a recorded "
        "artifact but have the test write only on a substantive change (compare before writing), "
        "or write the transcript to a temp path and assert on it there, leaving the committed copy "
        "as the checked-in reference; then un-dirty the working tree in CI."
    ),
    "evidence": (
        "git diff docs/evidence/hrs003/broken-case-transcript.txt after the control-host "
        "`cargo test --workspace` (rc=0) in this tick; tests/hrs003_python_interop.rs:23; "
        "commit d3877a8 (prior CI whitespace hit on the same path); "
        "docs/evidence/dogfood-20261017/control-host-cargo-test.log"
    ),
}
with open(BOARD, "a") as fh:
    fh.write(json.dumps(row, sort_keys=True) + "\n")
after = [json.loads(l)["id"] for l in open(BOARD).read().splitlines() if l.strip()]
assert after == before + [RID], "unexpected board id diff"
assert len(after) == len(set(after)), "duplicate ids"
print("appended", RID, "rows", len(before), "->", len(after))
