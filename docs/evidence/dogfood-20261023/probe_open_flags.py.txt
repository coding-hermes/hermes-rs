#!/usr/bin/env python3
"""Probe the adapter `open` boolean-typing contract (specs/python-interop.md §4.1).

Arms:
  A1 create:"yes"          -> spec promises InvalidInput naming params.create
  A2 create:1              -> same
  A3 read_only:"true"      -> spec promises InvalidInput naming params.read_only
  A4 read_only:true (bool) -> control: real read-only handle; append must fail
  A5 create:true  (bool)   -> control: creates
Prints one JSON object per arm.
"""
import json
import os
import subprocess
import sys

ADAPTER = sys.argv[1]
WORK = sys.argv[2]
os.makedirs(WORK, exist_ok=True)


class A:
    def __init__(self):
        self.p = subprocess.Popen([ADAPTER], stdin=subprocess.PIPE,
                                  stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                                  text=True, bufsize=1)
        self.i = 0

    def call(self, method, params=None):
        self.i += 1
        fr = {"id": self.i, "method": method}
        if params is not None:
            fr["params"] = params
        self.p.stdin.write(json.dumps(fr) + "\n")
        self.p.stdin.flush()
        line = self.p.stdout.readline()
        if not line:
            return {"_dead": self.p.poll(),
                    "_stderr": (self.p.stderr.read() or "").strip()[:400]}
        return json.loads(line)

    def kill(self):
        try:
            self.p.kill()
        except Exception:
            pass


def arm(name, params, then_append=None):
    path = os.path.join(WORK, f"arm-{name}.db")
    for suffix in ("", "-wal", "-shm"):
        try:
            os.unlink(path + suffix)
        except FileNotFoundError:
            pass
    # For read_only arms the file must exist first: create it with a boolean arm.
    if name.startswith("read_only"):
        ok = A()
        r = ok.call("open", {"path": path, "create": True})
        ok.call("ensure_session", {"session_id": "s1", "source": "probe"})
        ok.call("close")
        ok.kill()
    a = A()
    r = a.call("open", {"path": path, **params})
    res = {"arm": name, "params": params, "open_reply": r,
           "file_exists_after": os.path.exists(path)}
    if then_append:
        res["append_reply"] = a.call("append_message",
                                     {"session_id": "s1", "role": "user",
                                      "content": then_append})
    a.kill()
    res["file_exists_after"] = os.path.exists(path)
    res["bytes_after"] = os.path.getsize(path) if os.path.exists(path) else None
    print(json.dumps(res), flush=True)


arm("create_str_yes", {"create": "yes"})
arm("create_int_1", {"create": 1})
arm("read_only_str_true", {"read_only": "true"}, then_append="written-through-wrong-typed-flag")
arm("read_only_bool_true", {"read_only": True}, then_append="written-through-real-readonly")
arm("create_bool_true", {"create": True}, then_append="control-write")
