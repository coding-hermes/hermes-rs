#!/usr/bin/env python3
"""Protocol conformance: does `shutdown` reply before exiting, and with what rc?

Docs claim one reply line per request. Test: hello, shutdown, then drain stdout
for up to 5 s and report the exit code.

Usage: python3 probe_shutdown.py <adapter>
"""
import json
import subprocess
import sys
import time


def main():
    adapter = sys.argv[1]
    p = subprocess.Popen([adapter], stdin=subprocess.PIPE, stdout=subprocess.PIPE,
                         stderr=subprocess.PIPE, text=True, bufsize=1)
    p.stdin.write(json.dumps({"id": 1, "method": "hello"}) + "\n")
    p.stdin.flush()
    print("hello reply:", p.stdout.readline().strip()[:80])
    p.stdin.write(json.dumps({"id": 2, "method": "shutdown"}) + "\n")
    p.stdin.flush()
    t0 = time.time()
    line = p.stdout.readline()
    print(f"shutdown reply after {time.time()-t0:.2f}s: {line.strip()[:120]!r}")
    deadline = time.time() + 5
    while time.time() < deadline and p.poll() is None:
        time.sleep(0.1)
    print(f"exit after shutdown: rc={p.poll()} (waited {time.time()-t0:.2f}s)")
    if p.poll() is None:
        p.kill()
    print("stderr:", (p.stderr.read() or '').strip()[:200])


if __name__ == "__main__":
    main()
