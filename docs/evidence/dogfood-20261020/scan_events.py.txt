#!/usr/bin/env python3
"""Census the committed events.jsonl blob for fused/unparseable JSONL records."""
import json
import subprocess
import sys

repo = "/home/kara/hermes-rs"
blob = subprocess.run(
    ["git", "show", "HEAD:.coding-hermes/board/events.jsonl"],
    cwd=repo, capture_output=True, text=True, check=True,
).stdout
lines = blob.split("\n")
dec = json.JSONDecoder()
bad = []
for i, line in enumerate(lines, 1):
    if not line.strip():
        continue
    try:
        obj, idx = dec.raw_decode(line.strip())
    except Exception as e:
        bad.append((i, "unparseable", str(e)[:70]))
        continue
    rest = line.strip()[idx:].strip()
    if rest:
        bad.append((i, "fused", f"{len(rest)} trailing bytes starting {rest[:90]!r}"))
print(f"HEAD blob: {len(lines)} lines, {len(blob)} bytes")
print(f"fused/unparseable lines: {len(bad)}")
for b in bad:
    print("  ", b)
sys.exit(0)
