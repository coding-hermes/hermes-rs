#!/usr/bin/env python3
"""Cross-surface check: read a store WRITTEN by the Rust consumer through the
stdlib-Python adapter process (a different surface, different entry point).

Usage: python3 probe_adapter_read.py <adapter> <db> <session_id> <query>
"""
import json
import subprocess
import sys


def main():
    adapter, db, sid, query = sys.argv[1:5]
    frames = [
        {"id": 1, "method": "hello"},
        {"id": 2, "method": "open", "params": {"path": db, "create": False}},
        {"id": 3, "method": "get_session", "params": {"session_id": sid}},
        {"id": 4, "method": "search", "params": {"query": query, "limit": 5}},
        {"id": 5, "method": "get_messages", "params": {"session_id": sid, "limit": 2}},
        {"id": 6, "method": "shutdown"},
    ]
    p = subprocess.Popen([adapter], stdin=subprocess.PIPE, stdout=subprocess.PIPE,
                         stderr=subprocess.PIPE, text=True, bufsize=1)
    for f in frames:
        p.stdin.write(json.dumps(f) + "\n")
        p.stdin.flush()
        line = p.stdout.readline().strip()
        if not line:
            print(f"frame {f['method']}: <no reply; process rc={p.poll()}>")
            break
        obj = json.loads(line)
        if f["method"] == "open" and "result" in obj:
            r = obj["result"]
            print(f"open: capabilities={json.dumps(r.get('capabilities'))}")
            print(f"open: created_new_store={r['report'].get('created_new_store')} degraded={r['report'].get('degraded')}")
        elif f["method"] == "get_session":
            print(f"get_session: {json.dumps(obj)[:300]}")
        elif f["method"] == "search":
            res = obj.get("result", obj)
            hits = res.get("hits", res) if isinstance(res, dict) else res
            print(f"search over the wire: {len(hits) if isinstance(hits, list) else res} hits")
            if isinstance(hits, list) and hits:
                print(f"  first hit: {json.dumps(hits[0])[:200]}")
        elif f["method"] == "get_messages":
            res = obj.get("result", obj)
            msgs = res.get("messages", res) if isinstance(res, dict) else res
            print(f"get_messages over the wire: {len(msgs) if isinstance(msgs, list) else res}")
        else:
            print(f"{f['method']}: {json.dumps(obj)[:200]}")
    try:
        p.wait(timeout=10)
    except Exception:
        p.kill()


if __name__ == "__main__":
    main()
